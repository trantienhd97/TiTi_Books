    class ActiveChatTracker {
  String? _activeChannelCid;

  bool isActiveChannel(String? cid) {
    if (cid == null || cid.isEmpty) return false;
    return _activeChannelCid == cid;
  }

  void setActiveChannelCid(String? cid) {
    _activeChannelCid = cid;
  }
}
